-- НДС по операциям, справочно (лист «НДС по операциям»).
-- Строки: (me, kind, code, item, cost_item, cp, value), value — сумма НДС в рублях со знаком учёта.
--   sales — НДС с продаж WB: с цены покупателя, как в P&L (pl_sales.vat_amount);
--   comm  — НДС с вознаграждения WB по отчёту (как «Сверка НДС WB», стр. 2.2);
--   wb    — НДС с расходов и продвижения WB: расчётно по ставке строки;
--   gl    — НДС из учёта (gl.fact, chapter VAT_ACCURALS), разнесённый по счёту
--           группе статей, статье и контрагенту парной проводки P&L (тот же pid).
WITH sales_vat AS (
    SELECT LAST_DAY(date_from)::DATE AS me, 'sales' AS kind, '' AS code, '' AS item,
           '' AS cost_item, '' AS cp,
           SUM(vat_amount) AS value
    FROM pl_sales
    WHERE date_from::DATE <= $date_from
    GROUP BY 1
),
comm_vat AS (
    SELECT LAST_DAY(rr_dt::DATE)::DATE AS me, 'comm' AS kind, '' AS code, '' AS item,
           '' AS cost_item, '' AS cp,
           COALESCE(SUM(ppvz_vw_nds) FILTER (WHERE COALESCE(sop_name, '') = 'Продажа'), 0)
           - COALESCE(SUM(ppvz_vw_nds) FILTER (WHERE COALESCE(sop_name, '') = 'Возврат'), 0)
           + COALESCE(SUM(ppvz_vw_nds) FILTER (
                 WHERE COALESCE(sop_name, '') NOT IN ('Продажа', 'Возврат')), 0) AS value
    FROM sales.wb_raw
    WHERE rr_dt::DATE <= $date_from
    GROUP BY 1
),
wb_vat AS (
    SELECT me::DATE AS me, 'wb' AS kind,
           CASE WHEN account = 'WB Deduction' THEN 'promo' ELSE 'selling' END AS code,
           '' AS item, '' AS cost_item, '' AS cp,
           -SUM((dt - cr) * COALESCE(vat_rate, 20) / (100 + COALESCE(vat_rate, 20))) / 100
               AS value
    FROM wb_costs
    WHERE date_from::DATE <= $date_from
      AND account != 'Other income / loss'
    GROUP BY 1, 2, 3
),
-- НДС с выплат WB по программе сотрудничества: НДС внутри суммы, по ставке строки
wbpay_vat AS (
    SELECT me::DATE AS me, 'wbpay' AS kind, '' AS code, '' AS item,
           '' AS cost_item, '' AS cp,
           SUM((dt - cr) * COALESCE(vat_rate, 22) / (100 + COALESCE(vat_rate, 22))) / 100 AS value
    FROM wb_costs
    WHERE date_from::DATE <= $date_from AND cost_item = '5.4. Выплаты WB по программе сотрудничества'
    GROUP BY 1
),
gl_vat AS (
    SELECT LAST_DAY(v.date_from)::DATE AS me, 'gl' AS kind,
           COALESCE(LEFT(CAST(a.code AS VARCHAR), 3), '') AS code,
           COALESCE(NULLIF(TRIM(concat_ws(' ', sp.code, sp.name)), ''), 'Без группы') AS item,
           COALESCE(NULLIF(TRIM(concat_ws(' ', s.code, s.name)), ''), 'Без статьи') AS cost_item,
           COALESCE(NULLIF(TRIM(cp.name), ''), 'Без контрагента') AS cp,
           SUM(v.dt - v.cr) / 100.0 AS value
    FROM pg.gl.fact v
    -- статья и договор — из строки начисления (accural_distribution): при режиме
    -- vat_only проводки в P&L нет, а счёт P&L в начислении есть
    LEFT JOIN pg.gl.accural_distribution p ON p.id = v.pid
    LEFT JOIN pg.gl.fact pf ON pf.pid = v.pid AND pf.chapter = 'PL_ACCURALS'
    -- без счёта P&L (закупки на баланс: упаковка, материалы) — счёт BS
    LEFT JOIN pg.public.corporate_coa a ON a.id = COALESCE(p.acc_pl, pf.acc_id, p.acc_bs)
    LEFT JOIN pg.public.corporate_cfitems s ON s.id = COALESCE(p.subconto_pl, pf.subconto_id, p.subconto_bs)
    LEFT JOIN pg.public.corporate_cfitems sp ON sp.id = s.parent_id
    LEFT JOIN pg.public.contracts_contracts c ON c.id = COALESCE(p.contract_id, v.contract_id)
    LEFT JOIN pg.public.counterparties_counterparty cp ON cp.id = c.cp_id
    WHERE v.chapter = 'VAT_ACCURALS'
      AND v.date_from <= $date_from
    GROUP BY 1, 2, 3, 4, 5, 6
),
-- НДС по закупке товаров — из УПД поставщиков (upd_income_lines), по дате УПД.
-- goods_base — справочно сумма закупок без НДС в рублях, для сверки полноты.
-- Товар без УПД: НДС из назначения платежа (выписки, с учётом сплитов).
-- Только договоры с оплатой товара (статьи 121…), по которым нет ни одного УПД.
-- Сумма НДС из текста берётся, если сходится со ставкой из назначения (±5%) или
-- не больше 22/122 платежа; иначе — расчётно по ставке. Оплата — плюс (к вычету).
goods_cash_pay AS (
    SELECT t.date::DATE AS d, t.contract_id, t.cfitem_id, t.temp, t.vat_rate,
           COALESCE(t.cr, 0) - COALESCE(t.dt, 0) AS out_amt
    FROM pg.public.treasury_cfdata t
    UNION ALL
    SELECT t.date::DATE, s.contract_id, s.cfitem_id, COALESCE(s.temp, t.temp),
           COALESCE(s.vat_rate, t.vat_rate), COALESCE(s.cr, 0) - COALESCE(s.dt, 0)
    FROM pg.public.treasury_cfsplits s
    JOIN pg.public.treasury_cfdata t ON t.id = s.transaction_id
),
goods_cash_rows AS (
    SELECT p.d, p.contract_id, p.out_amt, p.vat_rate AS r,
           TRY_CAST(REPLACE(REPLACE(REPLACE(REGEXP_EXTRACT(p.temp,
               '(?i)ндс[^0-9]{0,20}(?:[0-9]{1,2}(?:[.,][0-9]+)?\s*%[^0-9]{0,15})?([0-9][0-9 ]{0,12}[.,-][0-9]{2})',
               1), ' ', ''), ',', '.'), '-', '.') AS DOUBLE) AS vat_txt,
           p.out_amt * COALESCE(p.vat_rate, 0) / (100 + COALESCE(p.vat_rate, 0)) AS vat_calc
    FROM goods_cash_pay p
    JOIN pg.public.corporate_cfitems i ON i.id = p.cfitem_id
    WHERE p.d <= $date_from AND p.out_amt <> 0 AND p.contract_id IS NOT NULL
      AND CAST(i.code AS VARCHAR) LIKE '121%'
      AND p.contract_id NOT IN (
          SELECT DISTINCT d.contract_id FROM pg.public.cards_upddocument d
          JOIN pg.public.upd_income_lines l ON l.upd_document_id = d.id
          WHERE d.contract_id IS NOT NULL)
),
goods_cash AS (
    SELECT d, contract_id,
           CASE WHEN vat_txt > 0 AND r > 0 AND ABS(vat_txt - ABS(vat_calc)) <= 0.05 * ABS(vat_calc) + 1
                    THEN SIGN(out_amt) * vat_txt
                WHEN vat_txt > 0 AND r IS NULL AND vat_txt <= ABS(out_amt) * 22 / 122 * 1.01
                    THEN SIGN(out_amt) * vat_txt
                WHEN r > 0 THEN vat_calc
                ELSE 0 END AS vat,
           CASE WHEN vat_txt > 0 AND ((r > 0 AND ABS(vat_txt - ABS(vat_calc)) <= 0.05 * ABS(vat_calc) + 1)
                                      OR (r IS NULL AND vat_txt <= ABS(out_amt) * 22 / 122 * 1.01))
                    THEN 'сумма из назначения платежа'
                ELSE 'расчётно по ставке из назначения' END AS src
    FROM goods_cash_rows
),
goods_cash_vat AS (
    SELECT LAST_DAY(g.d)::DATE AS me, 'goods_cash' AS kind, '' AS code,
           COALESCE(NULLIF(TRIM(cp.name), ''), 'Без поставщика') AS item,
           COALESCE(ct.title, 'Договор') || ' № ' || COALESCE(c.number, 'б/н') || ' · id '
               || CAST(c.id AS VARCHAR) AS cost_item,
           g.src AS cp, SUM(g.vat) AS value
    FROM goods_cash g
    JOIN pg.public.contracts_contracts c ON c.id = g.contract_id
    LEFT JOIN pg.public.contracts_contractstitle ct ON ct.id = c.title_id
    LEFT JOIN pg.public.counterparties_counterparty cp ON cp.id = c.cp_id
    WHERE g.vat <> 0
    GROUP BY 1, 2, 3, 4, 5, 6
),
goods_vat AS (
    SELECT LAST_DAY(d.date)::DATE AS me, 'goods' AS kind, '' AS code,
           CASE WHEN COALESCE(l.upd_vat_rate, 0) = 0 THEN 'НДС 0% (импорт / без НДС)'
                ELSE 'НДС ' || CAST(CAST(l.upd_vat_rate AS INTEGER) AS VARCHAR) || '%' END AS item,
           COALESCE(NULLIF(TRIM(cp.name), ''), 'Без поставщика') AS cost_item,
           'УПД № ' || d.number || ' от ' || strftime(d.date, '%d.%m.%Y') AS cp,
           SUM(COALESCE(l.upd_vat_amount,
                        l.upd_amount_vatless * COALESCE(l.upd_vat_rate, 0) / 100)) AS value
    FROM pg.public.upd_income_lines l
    JOIN pg.public.cards_upddocument d ON d.id = l.upd_document_id
    LEFT JOIN pg.public.counterparties_counterparty cp ON cp.id = d.counterparty_id
    WHERE d.date <= $date_from AND COALESCE(l.currency_code, 'RUB') = 'RUB'
    GROUP BY 1, 2, 3, 4, 5, 6
),
goods_base AS (
    SELECT LAST_DAY(d.date)::DATE AS me, 'goods_base' AS kind, '' AS code, '' AS item,
           '' AS cost_item, '' AS cp, SUM(l.upd_amount_vatless) AS value
    FROM pg.public.upd_income_lines l
    JOIN pg.public.cards_upddocument d ON d.id = l.upd_document_id
    WHERE d.date <= $date_from AND COALESCE(l.currency_code, 'RUB') = 'RUB'
    GROUP BY 1
)
SELECT * FROM sales_vat
UNION ALL SELECT * FROM goods_vat
UNION ALL SELECT * FROM goods_base
UNION ALL SELECT * FROM goods_cash_vat
UNION ALL SELECT * FROM comm_vat
UNION ALL SELECT * FROM wb_vat
UNION ALL SELECT * FROM wbpay_vat
UNION ALL SELECT * FROM gl_vat
