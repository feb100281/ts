-- ============================================================
-- ИСТОЧНИК УЧЁТНЫХ РАСХОДОВ ДЛЯ P&L (pl_src)
--
-- pl_for_csv (кассовый метод) + зарплата по начислению из ЗУП:
-- за месяцы, по которым есть расчётная ведомость ЗУП, оплата
-- труда (550100) из банка заменяется начислением; НДФЛ и
-- страховые взносы добавляются статьями в ту же группу
-- «Расходы на персонал». Месяцы без
-- ведомости остаются кассовыми. Подбор персонала и прочее — как было.
--
-- {payroll}, {contrib} — пути к data/zup/*.parquet (команда zup_payroll).
-- Используется в opex.txt, pivot_pl.txt, pl_notes.txt.
-- ============================================================

CREATE OR REPLACE TEMP VIEW pl_src AS

WITH zup_pay AS (
    SELECT * FROM read_parquet('{payroll}')
),

zup_con AS (
    SELECT * FROM read_parquet('{contrib}')
),

zup_months AS (
    SELECT DISTINCT period::DATE AS m FROM zup_pay
),

acc AS (
    SELECT
        COALESCE(MIN(account_name), '610000 Накладные расходы') AS account_name,
        COALESCE(MIN(cost_item_group) FILTER (WHERE STARTS_WITH(cost_item, '550100')),
                 '550000 Расходы на персонал') AS grp
    FROM pg.public.pl_for_csv
    WHERE STARTS_WITH(account_name, '610000')
),

cash AS (
    SELECT *
    FROM pg.public.pl_for_csv p
    WHERE NOT (
        STARTS_WITH(p.account_name, '610000')
        AND STARTS_WITH(COALESCE(p.cost_item, ''), '550100')
        AND DATE_TRUNC('month', p.date_from)::DATE IN (SELECT m FROM zup_months)
    )
),

accrual AS (
    SELECT
        z.period::DATE AS date_from,
        a.account_name,
        a.grp AS cost_item_group,
        '550100 Заработная плата по ведомости ЗУП (без НДФЛ)' AS cost_item,
        COALESCE(z.full_name, UPPER(z.name_short)) AS cp_name,
        'Начисление ЗУП' AS contract_name,
        -(z.accrued_total - z.ndfl)::DOUBLE AS amount,
        'ЗУП: расчётная ведомость' AS description
    FROM zup_pay z, acc a
    WHERE z.accrued_total - z.ndfl <> 0

    UNION ALL

    SELECT
        z.period::DATE,
        a.account_name,
        a.grp,
        '550100 НДФЛ по ведомости ЗУП',
        COALESCE(z.full_name, UPPER(z.name_short)),
        'Начисление ЗУП',
        -z.ndfl::DOUBLE,
        'ЗУП: расчётная ведомость'
    FROM zup_pay z, acc a
    WHERE z.ndfl <> 0

    UNION ALL

    SELECT
        c.period::DATE,
        a.account_name,
        a.grp,
        '550100 Страховые взносы: ' || CASE
            WHEN c.section ILIKE 'Единый тариф%' THEN 'единый тариф'
            WHEN c.section ILIKE '%несч%' THEN 'несчастные случаи и профзаболевания'
            ELSE c.section
        END,
        c.full_name,
        'Начисление ЗУП',
        -c.amount::DOUBLE,
        'ЗУП: проверка расчёта взносов'
    FROM zup_con c, acc a
    WHERE c.amount <> 0
)

SELECT * FROM cash
UNION ALL BY NAME
SELECT * FROM accrual
;
