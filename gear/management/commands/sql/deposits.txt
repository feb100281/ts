-- ============================================================
-- БЕССРОЧНЫЕ ДЕПОЗИТЫ НА ОТЧЁТНУЮ ДАТУ
--
-- acc_id = 10 — счёт бессрочных депозитов.
-- Сальдо считаем накопительно на дату отчёта.
-- Для актива: дебет увеличивает остаток, кредит уменьшает.
-- ============================================================

CREATE OR REPLACE TEMP TABLE deposits AS

SELECT
    ROUND(SUM(COALESCE(dt, 0) - COALESCE(cr, 0)) / 100.0, 0) AS balance

FROM pg.gl.fact

WHERE acc_id = 10
  AND date_from <= $date_from
;
